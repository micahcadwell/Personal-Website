  function fmt(s) {

    if (!isFinite(s)) return '—:——';

    const m = Math.floor(s / 60), sec = Math.floor(s % 60);

    return m + ':' + String(sec).padStart(2, '0');

  }


  function loadTrack(idx) {

    current = idx;

    audio.src = tracks[idx].src;

    trackTitle.textContent = tracks[idx].title;

    trackItems.forEach((el, i) => el.classList.toggle('active', i === idx));

    progFill.style.width = '0%';

    timeCur.textContent = '0:00';

    timeTotal.textContent = '—:——';

  function togglePlay() {

  audio.addEventListener('ended', () => { if (current < tracks.length - 1) { loadTrack(current + 1); audio.play().catch(() => {}); } else { isPlaying = false; btnPlay.textContent = '▶'; btnPlay.classList.remove('playing'); } });
